    81→/** Normalise an Indian mobile number to bare 10 digits (accepts +91 / 91 / 0 prefixes). */
    82→function normaliseMobile(raw: string): string {
    83→  const digits = raw.replace(/\D/g, "");
    84→  if (digits.length === 12 && digits.startsWith("91")) return digits.slice(2);
    85→  if (digits.length === 11 && digits.startsWith("0")) return digits.slice(1);
    86→  return digits;
    87→}
    88→
    89→function todayIso(): string {
    90→  const d = new Date();
    91→  const pad = (n: number) => String(n).padStart(2, "0");
    92→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
    93→}
    94→
    95→function FieldError({ id, message }: { id: string; message?: string }) {
    96→  if (!message) return null;
    97→  return (
    98→    <p id={id} role="alert" className="mt-1.5 text-xs font-medium text-destructive">
    99→      {message}
   100→    </p>
   101→  );
   102→}

   134→  function validate(): FormErrors {

   143→    if (form.message.length > 2000) errs.message = "Message is too long (max 2000 characters).";
   144→    if (!form.consent) errs.consent = "Please provide consent so our team can contact you.";
   145→    return errs;
   146→  }

   188→    }
