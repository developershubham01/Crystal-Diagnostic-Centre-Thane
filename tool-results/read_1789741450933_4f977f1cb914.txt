    84→    84→/** Normalise an Indian mobile number to bare 10 digits (accepts +91 / 91 / 0 prefixes). */
    85→    85→function normaliseMobile(raw: string): string {
    86→    86→  const digits = raw.replace(/\D/g, "");
    87→    87→  if (digits.length === 12 && digits.startsWith("91")) return digits.slice(2);
    88→    88→  if (digits.length === 11 && digits.startsWith("0")) return digits.slice(1);
    89→    89→  return digits;
    90→    90→}

    92→    92→function todayIso(): string {
    93→    93→  const d = new Date();
    94→    94→  const pad = (n: number) => String(n).padStart(2, "0");
    95→    95→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
    96→    96→}
