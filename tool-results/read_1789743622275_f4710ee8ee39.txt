    85→/** Normalise an Indian mobile number to bare 10 digits (accepts +91 / 91 / 0 prefixes). */
    86→function normaliseMobile(raw: string): string {
    87→  const digits = raw.replace(/\D/g, "");
    88→  if (digits.length === 12 && digits.startsWith("91")) return digits.slice(2);
    89→  if (digits.length === 11 && digits.startsWith("0")) return digits.slice(1);
    90→  return digits;
    91→}

    93→/** Today's date in the centre's timezone (Asia/Kolkata) as YYYY-MM-DD. */
    94→function todayIso(): string {
    95→  return new Intl.DateTimeFormat("en-CA", {
    96→    timeZone: "Asia/Kolkata",
    97→    year: "numeric",
    98→    month: "2-digit",
    99→    day: "2-digit",
   100→  }).format(new Date());
   101→}
   102→
   103→/** Weekday index (0 = Sunday) for a YYYY-MM-DD date, evaluated in IST. */
   104→function weekdayOf(dateIso: string): number {
   105→  return new Date(`${dateIso}T12:00:00+05:30`).getDay();
   106→}

   108→function FieldError({ id, message }: { id: string; message?: string }) {
   109→  if (!message) return null;

   111→    <p id={id} role="alert" className="mt-1.5 text-xs font-medium text-destructive">
   112→      {message}
   113→    </p>
   114→  );
   115→}
