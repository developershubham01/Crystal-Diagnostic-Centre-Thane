    81→    81→/** Normalise an Indian mobile number to bare 10 digits (accepts +91 / 91 / 0 prefixes). */
    82→    82→function normaliseMobile(raw: string): string {
    83→    83→  const digits = raw.replace(/\D/g, "");
    84→    84→  if (digits.length === 12 && digits.startsWith("91")) return digits.slice(2);
    85→    85→  if (digits.length === 11 && digits.startsWith("0")) return digits.slice(1);
    86→    86→  return digits;
    87→    87→}
    88→    88→
    89→    89→function todayIso(): string {
    90→    90→  const d = new Date();
    91→    91→  const pad = (n: number) => String(n).padStart(2, "0");
    92→    92→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
    93→    93→}

   102→   102→}
